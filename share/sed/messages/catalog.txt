# Message catalogue for the sed command-line stream editor.
#
# Format: locale.key = ICU message string, one message per line.
#
# This file is the authoritative source of every user-facing string. It is
# installed as <prefix>/share/sed/messages/catalog.txt and resolved relative to
# the executable, so the installed program never reads it through a source-tree
# or developer-local path. The same relative path works in the build tree,
# where bin/ and share/ are siblings of the crate root.
#
# Sed data, script text, option spellings, labels, file names and source
# excerpts are never translated: they appear only as arguments.

default_locale = en

# --- Diagnostic frames -----------------------------------------------------

en.sed.diagnostic.with_location = {program}: {location}: {severity}: {message}
en.sed.diagnostic.without_location = {program}: {severity}: {message}

en.sed.diagnostic.note = {program}: {location}: {severity}: {message}

en.sed.diagnostic.severity.error = error
en.sed.diagnostic.severity.warning = warning
en.sed.diagnostic.severity.information = note

# Source locations are machine-oriented, so their numbers are plain decimals
# rather than locale-grouped values.
en.sed.location.path = {path}

en.sed.location.expression = command line expression {index}
en.sed.location.expression_line = command line expression {index}:{line}
en.sed.location.expression_line_column = command line expression {index}:{line}:{column}

en.sed.help.options.heading = Options:
en.sed.option.quiet =   -n                     suppress automatic printing of the pattern space
en.sed.option.expression =   -e script              add the script text to the commands to run
en.sed.option.script_file =   -f script_file         add the contents of the file to the commands to run
en.sed.option.help =   --help                 show this help and exit
en.sed.option.version =   --version              show version information and exit
en.sed.option.color =   --color=auto|always|never  style diagnostics only, never sed output
en.sed.option.terminator =   --                     end option processing
en.sed.help.operands.heading = Operands:
en.sed.help.operands.script = Without -e and -f, the first operand is the script and the rest are input files.
en.sed.help.operands.files = All input operands form one logical stream: line numbers continue across files and the address $ matches only the last line of the whole stream.
en.sed.help.operands.stdin = An operand of - names standard input. With no input operand, standard input is read.
en.sed.help.exit.heading = Exit status:
en.sed.help.exit.success =   0  success, or --help or --version
en.sed.help.exit.processing =   1  script, input, output or execution failure
en.sed.help.exit.invocation =   2  invalid invocation
en.sed.help.exit.internal =   3  unexpected internal failure
en.sed.help.conformance.heading = Conformance:
en.sed.help.conformance.summary = This program targets POSIX sed. GNU extensions such as -i, -E, -r, -s, -u and -z are not implemented and are rejected as unknown options.
